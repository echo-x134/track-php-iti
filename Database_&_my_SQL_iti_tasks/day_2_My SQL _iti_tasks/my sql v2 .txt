CREATE TABLE Musician (
    ID INT PRIMARY KEY AUTO_INCREMENT,
    Name VARCHAR(100),
    Phone VARCHAR(20),
    Street VARCHAR(100),
    City VARCHAR(100)
);

CREATE TABLE Album (
    ID INT PRIMARY KEY AUTO_INCREMENT,
    Title VARCHAR(100),
    Cr_Date DATE,
    Musician_ID INT,
    FOREIGN KEY (Musician_ID) REFERENCES Musician(ID)
);

CREATE TABLE Song (
    Title VARCHAR(100) PRIMARY KEY,
    Author VARCHAR(100),
    Album_ID INT,
    FOREIGN KEY (Album_ID) REFERENCES Album(ID)
);

);

CREATE TABLE Play_Instrument (
    Musician_ID INT,
    Instrument_Name VARCHAR(100),
    PRIMARY KEY (Musician_ID, Instrument_Name),
    FOREIGN KEY (Musician_ID) REFERENCES Musician(ID),
    FOREIGN KEY (Instrument_Name) REFERENCES Instrument(Name)
);

============================================================================
CREATE TABLE Play_Song (
    Musician_ID INT,
    Song_Title VARCHAR(100),
    PRIMARY KEY (Musician_ID, Song_Title),
    FOREIGN KEY (Musician_ID) REFERENCES Musician(ID),
    FOREIGN KEY (Song_Title) REFERENCES Song(Title)
